-- Movie table
CREATE TABLE Movie (
    movie_id      NUMBER PRIMARY KEY,
    movie_name    VARCHAR2(15),
    genre         VARCHAR2(10),
    release_date  DATE
);
-- Screen table
CREATE TABLE Screen (
    screen_id     NUMBER PRIMARY KEY,
    screen_name   VARCHAR2(12),
    max_capacity  NUMBER
);
-- Current_Movie table (movie running info)
CREATE TABLE Current_Movie (
    movie_id        NUMBER,
    screen_id       NUMBER,
    date_of_arrival DATE,
    date_of_closure DATE,
    CONSTRAINT fk_movie FOREIGN KEY (movie_id) REFERENCES Movie(movie_id),
    CONSTRAINT fk_screen FOREIGN KEY (screen_id) REFERENCES Screen(screen_id)
);
