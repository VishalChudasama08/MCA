  2      -- input
  3      p_name VARCHAR2(20);
  4      -- supplier
  5      v_sid   suppliers.s_id%TYPE;
  6      v_sname suppliers.s_name%TYPE;
  7      -- catalog
  8      c_pid   catalog.part_id%TYPE;
  9      c_qty   catalog.qty%TYPE;
 10      -- parts
 11      p_pid   parts.part_id%TYPE;
 12      p_pname parts.part_name%TYPE;
 13      p_rate  parts.unit_rate%TYPE;
 14      v_total NUMBER;
 15      v_grand NUMBER := 0;
 16      supplier_not_found EXCEPTION;
 17      -- supplier cursor
 18      CURSOR cur_supplier IS
 19          SELECT s_id, s_name
 20          FROM suppliers
 21          WHERE s_name = p_name;
 22      -- catalog cursor (depends on supplier)
 23      CURSOR cur_catalog IS
 24          SELECT part_id, qty
 25          FROM catalog
 26          WHERE s_id = v_sid;
 27      -- parts cursor
 28      CURSOR cur_parts IS
 29          SELECT part_id, part_name, unit_rate
 30          FROM parts;
 31  BEGIN
 32      -- take input inside BEGIN
 33      p_name := '&p_name';
 34      -- check supplier
 35      OPEN cur_supplier;
 36      FETCH cur_supplier INTO v_sid, v_sname;
 37      IF cur_supplier%NOTFOUND THEN
 38          RAISE supplier_not_found;
 39      END IF;
 40      CLOSE cur_supplier;
 41      -- report header
 42      DBMS_OUTPUT.PUT_LINE('***********************************************************');
 43      DBMS_OUTPUT.PUT_LINE('PART ID   PART NAME     QUANTITY   UNIT PRICE   TOTAL');
 44      DBMS_OUTPUT.PUT_LINE('***********************************************************');
 45      -- catalog loop
 46      OPEN cur_catalog;
 47      LOOP
 48          FETCH cur_catalog INTO c_pid, c_qty;
 49          EXIT WHEN cur_catalog%NOTFOUND;
 50          -- parts loop
 51          OPEN cur_parts;
 52          LOOP
 53              FETCH cur_parts INTO p_pid, p_pname, p_rate;
 54              EXIT WHEN cur_parts%NOTFOUND;
 55              IF c_pid = p_pid THEN
 56                  v_total := c_qty * p_rate;
 57                  v_grand := v_grand + v_total;
 58                  DBMS_OUTPUT.PUT_LINE(
 59                      RPAD(p_pid,8) ||
 60                      RPAD(p_pname,14) ||
 61                      RPAD(c_qty,11) ||
 62                      RPAD(p_rate,13) ||
 63                      v_total
 64                  );
 65              END IF;
 66          END LOOP;
 67          CLOSE cur_parts;
 68      END LOOP;
 69      CLOSE cur_catalog;
 70      DBMS_OUTPUT.PUT_LINE('***********************************************************');
 71      DBMS_OUTPUT.PUT_LINE('Grand Total: ' || v_grand);
 72  EXCEPTION
 73      WHEN supplier_not_found THEN
 74          DBMS_OUTPUT.PUT_LINE('Error: Supplier name not found.');
 75* END;
